    def do_GET(self):
        """Handle GET requests"""
        parsed_url = urlparse(self.path)
        path = parsed_url.path
        query_params = parse_qs(parsed_url.query)
        
        print(f"GET {self.path}")
        
        # Moonraker API endpoints
        if path == '/server/info':
            self.send_server_info()
        elif path == '/printer/info':
            self.send_printer_info()
        elif path == '/printer/objects/list':
            self.send_objects_list()
        elif path == '/printer/objects/query':
            self.send_objects_query(query_params)
        elif path == '/server/files/list':
            self.send_files_list(query_params)
        elif path == '/server/temperature_store':
            self.send_temperature_store()
        # Printer Buddy API endpoints (proxy to mock data)
        elif path == '/api/status':
            self.send_printer_buddy_status()
        elif path == '/api/config':
            self.send_printer_buddy_config()
        elif path == '/api/logs/list':
            self.send_logs_list()
        elif path == '/api/logs/file':
            self.send_log_file(query_params) 
        elif path == '/api/logs':
            self.send_printer_buddy_logs()
        elif path == '/api/tests/list':
            self.send_tests_list()
        elif path == '/api/tests/status':
            self.send_test_status(query_params)
        # Serve static files for UI
        elif path == '/' or path == '/index.html':
            self.serve_ui_file('index.html')
        elif path.startswith('/css/'):
            self.serve_ui_file(path[1:])
        elif path.startswith('/js/'):
            self.serve_ui_file(path[1:])
        else:
            self.send_404()
    
    def do_POST(self):
        """Handle POST requests"""
        parsed_url = urlparse(self.path)
        path = parsed_url.path
        
        print(f"POST {self.path}")

            self.send_gcode_script()

            self.send_api_gcode()

            self.handle_test_run()
        elif path == '/api/tests/respond':
            self.handle_test_respond()
        elif path == '/printer/print/start':
            self.send_print_control('start')
        elif path == '/printer/print/pause':
            self.send_print_control('pause')
        elif path == '/printer/print/resume':
            self.send_print_control('resume')
        elif path == '/printer/print/cancel':
            self.send_print_control('cancel')
        else:
            self.send_404()
    
    def run_async(self, coro):
        """Helper to run async functions in sync context"""
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        try:
            return loop.run_until_complete(coro)
        finally:
            loop.close()

    def handle_test_run(self):
        """Handle test run POST request"""
        content_length = int(self.headers.get('Content-Length', 0))
        post_data = self.rfile.read(content_length).decode('utf-8')
        
        print(f"DEBUG: Received POST data: {post_data}")
        
        try:
            data = json.loads(post_data)
            test_id = data.get('test_id', '')
            
            print(f"DEBUG: Parsed test_id: '{test_id}'")
            
            if not test_id:
                response = {
                    "error": {
                        "code": 400,
                        "message": "test_id is required"
                    }
                }
                print(f"DEBUG: Missing test_id, sending 400 error")
                self.send_json_response(response, 400)
                return
            
            print(f"DEBUG: About to run test: {test_id}")
            # Run async method
            self.run_async(self.send_test_run(test_id))
            
        except json.JSONDecodeError:
            response = {
                "error": {
                    "code": 400,
                    "message": "Invalid JSON in request body"
                }
            }
            self.send_json_response(response, 400)
        except Exception as e:
            response = {
                "error": {
                    "code": 500,
                    "message": f"Internal error: {str(e)}"
                }
            }
            self.send_json_response(response, 500)

    def handle_test_respond(self):
        """Handle test response POST request"""
        content_length = int(self.headers.get('Content-Length', 0))
        post_data = self.rfile.read(content_length).decode('utf-8')
        
        try:
            data = json.loads(post_data)
            run_id = data.get('run_id', '')
            
            if not run_id:
                response = {
                    "error": {
                        "code": 400,
                        "message": "run_id is required"
                    }
                }
                self.send_json_response(response, 400)
                return
            
            # Run async method
            self.run_async(self.send_test_respond(run_id, data))
            
        except json.JSONDecodeError:
            response = {
                "error": {
                    "code": 400,
                    "message": "Invalid JSON in request body"
                }
            }
            self.send_json_response(response, 400)
        except Exception as e:
            response = {
                "error": {
                    "code": 500,
                    "message": f"Internal error: {str(e)}"
                }
            }
            self.send_json_response(response, 500)

    def send_json_response(self, data, status=200):
        """Send JSON response with proper headers"""
        self.send_response(status)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        self.end_headers()
        
        json_data = json.dumps(data, indent=2)
        self.wfile.write(json_data.encode())

    def send_api_gcode(self):
        """Handle G-code commands sent through API"""
        content_length = int(self.headers.get('Content-Length', 0))
        post_data = self.rfile.read(content_length).decode('utf-8')
        
        try:
            data = json.loads(post_data)
            command = data.get('command', '')
            
            print(f"G-code command received: {command}")
            
            # Mock response based on command
            if command.upper().startswith('M104') or command.upper().startswith('M109'):
                response = {
                    "success": True,
                    "message": f"Heating extruder..."
                }
            elif command.upper().startswith('M140') or command.upper().startswith('M190'):
                response = {
                    "success": True,
                    "message": f"Heating bed..."
                }
            elif command.upper().startswith('G28'):
                response = {
                    "success": True,
                    "message": f"Homing axes..."
                }
            elif command.upper().startswith('M112'):
                response = {
                    "success": True,
                    "message": f"Emergency stop triggered!"
                }
            else:
                response = {
                    "success": True,
                    "message": f"Command '{command}' executed"
                }
            
        except json.JSONDecodeError:
            response = {
                "success": False,
                "error": "Invalid JSON in request body"
            }
        
        self.send_json_response(response)
    
    def serve_ui_file(self, file_path):
        """Serve static UI files"""
        # Look for UI files in the web directory
        web_dir = os.path.join(os.path.dirname(os.path.dirname(__file__)), '..', 'web')
        full_path = os.path.join(web_dir, file_path)
        
        if os.path.exists(full_path):
            try:
                with open(full_path, 'rb') as f:
                    content = f.read()
                
                # Determine content type
                if file_path.endswith('.html'):
                    content_type = 'text/html'
                elif file_path.endswith('.css'):
                    content_type = 'text/css'
                elif file_path.endswith('.js'):
                    content_type = 'application/javascript'
                else:
                    content_type = 'text/plain'
                
                self.send_response(200)
                self.send_header('Content-Type', content_type)
                self.send_header('Access-Control-Allow-Origin', '*')
                self.end_headers()
                self.wfile.write(content)
                
            except Exception as e:
                print(f"Error serving {file_path}: {e}")
                self.send_404()
        else:
            print(f"File not found: {full_path}")
            self.send_404()

class ThreadedHTTPServer(socketserver.ThreadingMixIn, HTTPServer):
    """Threaded HTTP server for handling multiple connections"""
    allow_reuse_address = True

def start_mock_server(port=7125, host='localhost'):
    """Start the mock Moonraker server"""
    server_address = (host, port)
